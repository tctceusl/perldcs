##04 -- Calculator

	.data 
    msg1    : .asciiz "Enter the Number 1 : "
    msg2    : .asciiz "Enter the Number 2 : " 
    menu    : .asciiz "\n   Menu:\n   1- Addition\n   2- Sub\n   3- Mul\n   4- Div\n   5- Mod\n"
    sellMsg : .asciiz "Select the Operation from the Menu : "
    addOp   : .asciiz " + "
    subOp   : .asciiz " - "          
    mulOp   : .asciiz " x "          
    divOp   : .asciiz " / "          
    modOp   : .asciiz " % "          
    equalsOp: .asciiz " = "
    newline : .asciiz "\n"            
    divZero : .asciiz "Error: division by zero\n"   

.text 
    .global main 
    main:
        la $a0, msg1
        jal prompt_int 
        move $s0, $v0
        
        la $a0, msg2 
        jal prompt_int
        move $s1, $v0
        
        la $a0, menu 
        jal print_string
        
        la $a0, sellMsg 
        jal prompt_int
        move $t0, $v0 
        
        li $t1, 1
        beq $t0, $t1, do_add
        li $t1, 2 
        beq $t0, $t1, do_sub
        li $t1, 3 
        beq $t0, $t1, do_mul
        li $t1, 4 
        beq $t0, $t1, do_div
        li $t1, 5
        beq $t0, $t1, do_mod
        j end 
        
    do_add: 
        add $s2, $s0, $s1 
        move $a0, $s0 
        jal print_int 
        la $a0, addOp
        jal print_string 
        move $a0, $s1
        jal print_int                 
        la $a0, equalsOp
        jal print_string
        move $a0, $s2 
        jal print_int
        j end 
        
    do_sub: 
        sub $s2, $s0, $s1 
        move $a0, $s0 
        jal print_int 
        la $a0, subOp                 
        jal print_string 
        move $a0, $s1
        jal print_int                 
        la $a0, equalsOp
        jal print_string
        move $a0, $s2 
        jal print_int
        j end 
        
    do_mul: 
        mul $s2, $s0, $s1 
        move $a0, $s0 
        jal print_int 
        la $a0, mulOp                 
        jal print_string 
        move $a0, $s1
        jal print_int                 
        la $a0, equalsOp
        jal print_string
        move $a0, $s2 
        jal print_int
        j end 

    do_div: 
        beqz $s1, div_by_zero 
        div $s0, $s1
        mflo $s2 
        move $a0, $s0 
        jal print_int 
        la $a0, divOp                 
        jal print_string 
        move $a0, $s1 
        jal print_int 
        la $a0, equalsOp
        jal print_string 
        move $a0, $s2 
        jal print_int 
        j end  
        
    do_mod: 
        beqz $s1, div_by_zero 
        div $s0, $s1
        mfhi $s2 
        move $a0, $s0 
        jal print_int 
        la $a0, modOp                 
        jal print_string 
        move $a0, $s1 
        jal print_int 
        la $a0, equalsOp
        jal print_string 
        move $a0, $s2 
        jal print_int 
        j end 

    div_by_zero: 
        la $a0, divZero 
        jal print_string 
        
    end: 
        la $a0, newline 
        jal print_string
        jal exit_program


    # exit 
    exit_program: 
        li $v0, 10 
        syscall 

    # print string 
    print_string: 
        li $v0, 4 
        syscall
        jr $ra

    # prompt int (print prompt then read integer)
    prompt_int:
        li $v0, 4 
        syscall 
        li $v0, 5
        syscall 
        jr $ra 
        
    # print int 
    print_int: 
        li $v0, 1
        syscall 
        jr $ra

	print_array : 
		la $a0 , printMsg 
		li $v0 , 4
		syscall 
		
		la $t0 , arry
		lw $t1 , size 
		li $t2 , 0
		
	print_loop : 
		beq $t2 , $t1 , print_done 
		lw $a0 , 0($t0) 
		li $v0 , 1
		syscall 
		
		la $a0 , space

		addi $t0 , $t0 ,4 
		addi $t2 , $t2 , 1

	find_max : 
		lw  $t1 , 0($a0)

		addi $a0 , $a0 ,4 
		
	max_loop : 
		beq $t2 , $a1 , max_done 
		
		lw $t3 , 0($a0)
		ble $t3 , $t1 , max_skip 
		move $t1  , $t3 
		
	max_skip :
		addi $a0 , $a0 ,4 
		addi $t2  , $t2 ,1
		j max_loop
		
	max_done  :
		move $v0 , $t1 
		jr $ra
		
	
	find_sum :
		li $t1 , 0 
		li $t2 , 0 
		
	sum_loop : 
		beq $t2 , $a1 , sum_done 
		
		lw $t3 , 0($a0) 
		add $t1 , $t1 , $t3 
		
		addi $a0 , $a0 , 4 
		addi $t2 , $t2 , 1
		j sum_loop

		jr $ra
