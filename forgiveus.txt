        jal  exit_program

# a) exit from program
exit_program:
    li   $v0, 10
    syscall

# b) print a String (string address in $a0)
print_string:
    li   $v0, 4
    syscall
    jr   $ra

# c) prompt and read an Integer (prompt string in $a0, returns int in $v0)
prompt_int:
    li   $v0, 4
    syscall
    li   $v0, 5
    syscall
    jr   $ra

# d) print an Integer (integer in $a0)
print_int:
    li   $v0, 1
    syscall
    jr   $ra

# e) extract a digit from an Integer
# Input : $a0 = number, $a1 = digit position (1=units, 2=tens, 3=hundreds)
# Output: $v0 = extracted digit
extract_digit:
    li   $t3, 1
    extract_loop:
        beq  $a1, $t3, do_extract
        li   $t4, 10
        div  $a0, $t4
        mflo $a0
        addi $t3, $t3, 1
        j    extract_loop
    do_extract:
        li   $t4, 10
        div  $a0, $t4
        mfhi $v0             # remainder = digit
        jr   $ra







 # Armstrong number: N = d1^3 + d2^3 + d3^3
# Test with: 153, 125

.data
    inputMsg : .asciiz "Enter a 3-digit integer: "
    yesMsg   : .asciiz " is an Armstrong number!\n"
    noMsg    : .asciiz " is NOT an Armstrong number.\n"
    newline  : .asciiz "\n"

.text
    .globl main
    main:

    # --- Read number ---
    la   $a0, inputMsg
    jal  prompt_int
    move $s0, $v0            # $s0 = N

    # --- Extract hundreds digit (d1) ---
    move $a0, $s0
    li   $a1, 3              # position 3 = hundreds
    jal  extract_digit
    move $t0, $v0            # $t0 = d1
    mul  $t0, $t0, $t0
    mul  $t0, $t0, $v0       # Wait — recalculate properly below

    # Re-extract cleanly
    # d3 = N % 10
    li   $t4, 10
    div  $s0, $t4
    mfhi $t2                 # $t2 = d3 (units)
    mflo $t5                 # $t5 = N/10

    # d2 = (N/10) % 10
    div  $t5, $t4
    mfhi $t1                 # $t1 = d2 (tens)
    mflo $t5                 # $t5 = N/100

    # d1 = (N/100) % 10
    div  $t5, $t4
    mfhi $t0                 # $t0 = d1 (hundreds)

    # --- Calculate d1^3 + d2^3 + d3^3 ---
    mul  $s1, $t0, $t0
    mul  $s1, $s1, $t0       # $s1 = d1^3

    mul  $s2, $t1, $t1
    mul  $s2, $s2, $t1       # $s2 = d2^3

    mul  $s3, $t2, $t2
    mul  $s3, $s3, $t2       # $s3 = d3^3

    add  $s4, $s1, $s2
    add  $s4, $s4, $s3       # $s4 = sum of cubes

    # --- Compare with N ---
    move $a0, $s0
    jal  print_int

    beq  $s4, $s0, is_armstrong
    j    not_armstrong

    is_armstrong:
        la   $a0, yesMsg
        jal  print_string
        j    end

    not_armstrong:
        la   $a0, noMsg
        jal  print_string

    end:
    jal  exit_program

# Array: 56, 27, 13, 5, 44, 32, 17, 36
# Prompt user for integer, search, print found/not found

.data
    arr      : .word 56, 27, 13, 5, 44, 32, 17, 36
    arr_size : .word 8
    inputMsg : .asciiz "Enter an integer to search: "
    foundMsg : .asciiz "Integer is found.\n"
    notFound : .asciiz "Integer is not found.\n"

.text
    .globl main
    main:

    # --- Read search target ---
    la   $a0, inputMsg
    jal  prompt_int
    move $s0, $v0            # $s0 = target

    # --- Setup loop ---
    la   $t0, arr            # $t0 = base address of array
    lw   $t1, arr_size       # $t1 = size (8)
    li   $t2, 0              # $t2 = index counter
    li   $s1, 0              # $s1 = found flag (0=not found)

    search_loop:
        beq  $t2, $t1, search_done   # if index == size, stop

        lw   $t3, 0($t0)             # load arr[i]
        beq  $t3, $s0, found         # if arr[i] == target, found

        addi $t0, $t0, 4             # move to next element
        addi $t2, $t2, 1             # i++
        j    search_loop

    found:
        li   $s1, 1                  # set found flag

    search_done:

    # --- Print result ---
    beqz $s1, print_not_found

    la   $a0, foundMsg
    jal  print_string
    j    end

    print_not_found:
    la   $a0, notFound
    jal  print_string

    end:
    jal  exit_program

# --- Utility subprograms ---
exit_program:
